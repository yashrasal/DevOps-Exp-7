1	docker --version	    Check Docker version
2	docker info	            Check Docker Engine
3	mkdir java-docker-app	Create project folder
4	cd java-docker-app	    Enter project folder
5	HelloDocker.java	    Create Java file
6	Dockerfile	            Create Dockerfile
7	docker build -t java-docker-app:1.0 .	Build Docker Image
8	docker images	List Docker Images
9	docker run java-docker-app:1.0	        Run Docker Container
10	docker ps	                           Show running containers
11	docker ps -a	                        Show all containers
